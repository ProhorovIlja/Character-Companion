cmake_minimum_required(VERSION 3.31.6)
project(mandy_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Standalone speech preview requires Android ARM64")
endif()
find_package(Git REQUIRED)
if(NOT IS_DIRECTORY "${MANDY_WHISPER_DIR}")
  message(FATAL_ERROR "Provide an explicitly prepared whisper.cpp checkout")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MANDY_WHISPER_DIR}" rev-parse HEAD
  OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE rc)
if(NOT rc EQUAL 0 OR NOT actual STREQUAL "c1a24047871444999b79f5e8612828f45881ae12")
  message(FATAL_ERROR "Unexpected whisper.cpp revision")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MANDY_WHISPER_DIR}" status --porcelain --untracked-files=all
  OUTPUT_VARIABLE changes OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE rc)
if(NOT rc EQUAL 0 OR NOT changes STREQUAL "")
  message(FATAL_ERROR "whisper.cpp checkout must be clean")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_USE_SYSTEM_GGML OFF CACHE BOOL "" FORCE)
set(WHISPER_USE_SYSTEM_LLAMA OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(WHISPER_SDL2 OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${MANDY_WHISPER_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/whisper")
add_library(mandy_whisper SHARED mandy_whisper.cpp)
target_link_libraries(mandy_whisper PRIVATE whisper)
target_link_options(mandy_whisper PRIVATE
  "-Wl,-z,max-page-size=16384"
  "-Wl,--exclude-libs,ALL"
  "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
set_property(TARGET mandy_whisper APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
